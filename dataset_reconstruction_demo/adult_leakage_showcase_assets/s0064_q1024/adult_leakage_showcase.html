FROM tabular_data

代表性查询

SELECT workclass, fnlwgt, education, educational-num, marital-status, occupation, relationship, race, gender, capital-gain, capital-loss, native-country, income
FROM tabular_data

  AND hours-per-week BETWEEN 1 AND 99;
